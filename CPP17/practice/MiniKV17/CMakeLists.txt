cmake_minimum_required(VERSION 3.16)
project(MiniKV17 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MINIKV17_PARALLEL "Use par for snapshot statistics (backend dependent)" OFF)
find_package(Threads REQUIRED)
add_library(minikv17 src/model.cpp src/journal.cpp src/store.cpp src/shell.cpp)
target_include_directories(minikv17 PUBLIC include)
target_link_libraries(minikv17 PUBLIC Threads::Threads)
if(MINIKV17_PARALLEL)
    find_package(TBB REQUIRED)
    target_link_libraries(minikv17 PUBLIC TBB::tbb)
    target_compile_definitions(minikv17 PRIVATE MINIKV17_PARALLEL=1)
endif()
add_executable(minikv src/main.cpp)
target_link_libraries(minikv PRIVATE minikv17)
add_executable(minikv_tests tests/self_test.cpp)
target_link_libraries(minikv_tests PRIVATE minikv17)
foreach(target minikv17 minikv minikv_tests)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors)
    endif()
endforeach()
enable_testing()
add_test(NAME minikv_tests COMMAND minikv_tests)
add_test(NAME minikv_help COMMAND minikv --help)
set_tests_properties(minikv_tests minikv_help PROPERTIES TIMEOUT 30)
