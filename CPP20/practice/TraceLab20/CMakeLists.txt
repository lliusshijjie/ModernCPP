cmake_minimum_required(VERSION 3.22)
project(TraceLab20 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
option(TRACELAB_MODULES "Build the CLI that imports tracelab.model (CMake 3.28+)" OFF)
find_package(Threads REQUIRED)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

add_library(trace_core src/input.cpp src/analysis.cpp src/session.cpp src/report.cpp src/snapshot.cpp)
target_include_directories(trace_core PUBLIC include)
target_link_libraries(trace_core PUBLIC Threads::Threads)
add_library(trace_cli src/cli.cpp)
target_link_libraries(trace_cli PUBLIC trace_core)

function(trace_app name entry)
  add_executable(${name} ${entry})
  target_link_libraries(${name} PRIVATE trace_cli)
  if(MINGW)
    target_link_options(${name} PRIVATE -municode)
  endif()
endfunction()
trace_app(tracelab src/main.cpp)

if(TRACELAB_MODULES)
  if(CMAKE_VERSION VERSION_LESS 3.28)
    message(FATAL_ERROR "TRACELAB_MODULES requires CMake 3.28 or later")
  endif()
  trace_app(tracelab_modules modules/main.cpp)
  target_sources(tracelab_modules PRIVATE FILE_SET CXX_MODULES FILES modules/tracelab.model.cppm)
  set_property(TARGET tracelab_modules PROPERTY CXX_SCAN_FOR_MODULES ON)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(tracelab_tests tests/self_test.cpp)
  target_link_libraries(tracelab_tests PRIVATE trace_core)
  foreach(group input analysis queue snapshots zones)
    add_test(NAME ${group} COMMAND tracelab_tests ${group})
  endforeach()
  try_compile(accepted_duplicate "${CMAKE_CURRENT_BINARY_DIR}/check_duplicate"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/compile_fail/duplicate.cpp" CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON)
  if(accepted_duplicate)
    message(FATAL_ERROR "schema accepted a duplicate field")
  endif()
  foreach(app tracelab tracelab_modules)
    if(TARGET ${app})
      add_test(NAME cli_${app} COMMAND ${CMAKE_COMMAND}
        -DAPP=$<TARGET_FILE:${app}> -DSAMPLES=${CMAKE_CURRENT_SOURCE_DIR}/data
        -DOUT=${CMAKE_CURRENT_BINARY_DIR}/test-${app} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake)
    endif()
  endforeach()
  get_property(checks DIRECTORY PROPERTY TESTS)
  set_tests_properties(${checks} PROPERTIES TIMEOUT 45)
endif()
